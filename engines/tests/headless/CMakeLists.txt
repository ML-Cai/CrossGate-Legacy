cmake_minimum_required(VERSION 3.21)
project(CrossGateHeadlessTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(CGL_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${CGL_REPOSITORY_ROOT}/cmake/AddressSanitizer.cmake")
if(MSVC)
    add_compile_options(/EHsc /utf-8)
endif()
# Only this pinned source dependency is required; no desktop dependency setup.
if(NOT EXISTS "${CGL_REPOSITORY_ROOT}/third_party/googletest/CMakeLists.txt")
    message(FATAL_ERROR "Initialize only googletest: git submodule update --init third_party/googletest")
endif()
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${CGL_REPOSITORY_ROOT}/third_party/googletest" googletest EXCLUDE_FROM_ALL)
enable_testing()
include("${CGL_REPOSITORY_ROOT}/cmake/HeadlessCoreTests.cmake")
cgl_add_headless_core_tests("${CGL_REPOSITORY_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/bin")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
foreach(tool IN ITEMS StreamingStress LifecycleSoak)
    if(tool STREQUAL "StreamingStress")
        set(script test_stress_runner.py)
    else()
        set(script test_runner.py)
    endif()
    add_test(NAME ${tool}_protocol COMMAND ${Python3_EXECUTABLE} "${CGL_REPOSITORY_ROOT}/tools/${tool}/${script}")
    set_tests_properties(${tool}_protocol PROPERTIES LABELS "unit;memory;python;headless" TIMEOUT 30)
endforeach()

add_test(NAME anime_viewer_selection
    COMMAND ${Python3_EXECUTABLE} "${CGL_REPOSITORY_ROOT}/tools/AnimeViewer/tests/test_preview.py")
set_tests_properties(anime_viewer_selection PROPERTIES LABELS "unit;memory;python;headless" TIMEOUT 30)
